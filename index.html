function makeHaengbalStudentList() { const select = document.getElementById("haengbalStudentSelect"); if (!select) return; // 초기화 select.innerHTML = ``; // API 데이터가 아직 없으면 종료 if ( !Array.isArray(haengbalData) || haengbalData.length === 0 ) { select.innerHTML = ``; return; } const students = {}; haengbalData.forEach(row => { const number = String(row["학번"] || "").trim(); const name = String(row["이름"] || "").trim(); if (number) { students[number] = name; } }); // 학생번호 정렬 const sortedNumbers = Object.keys(students).sort((a, b) => { const aa = a.split("-").map(x => Number(x)); const bb = b.split("-").map(x => Number(x)); for ( let i = 0; i < Math.max(aa.length, bb.length); i++ ) { const av = aa[i] || 0; const bv = bb[i] || 0; if (av !== bv) { return av - bv; } } return 0; }); // 학생 목록 생성 sortedNumbers.forEach(number => { const option = document.createElement("option"); option.value = number; option.textContent = number + " " + (students[number] || ""); select.appendChild(option); }); }
